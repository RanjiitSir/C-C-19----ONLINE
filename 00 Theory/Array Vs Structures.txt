
Array =>

	Array is Derived Data Type in C

	Collection of HomoGeneous(Same) DataType Elements,
	Under a Single Name

	Sequential Byte Of memory 

	Array Indexing Starts with 0, Last Index is (ArraySize - 1)

==================================================

Structure =>
	
	Structure is User(Programmer) Defined Datatype in C

	Collection of Non-Homogeneous(Different) DataType Elements,
	Under a Single Name

	Sequential Byte Of memory 

	Sequence Of Declaration of elements in Structure
	in that sequence memory gets allocated 

	sizeof of Structure object = Summation of its all 
			              data members (Variables) in Structure 

==================================================

	We Can't Initialize Any Structure member while declaration,
	because at the time of declaration there is No Memory Allocated 
	for those members
	
	
	int No = 15;
	
	struct Cake
	{
			int CID;
			char OrderName[20];
			int MobileNo;
			float Amount;
			float weight;
			char Grade;
	};

	struct Cake Ck1 = {21, "Harry Potter", 984654645, 500.0,1000, 'B'};
	
	strcpy(Ck1.OrderName, "Harry Potter");
	
	
	struct Cake Ck[5];
	
	int Num = 11;
	
==================================================
Variable Declaration

	DataType <VarName>;
	
Exa:
	int No;
	
	char ch;
	
	float Per;
	
	
=================================================
Variable Initialization:
	DataType <VarName> = <InitialialValue>;

Exa:
	int No = 0;
	
	char ch = '\0';
	
	float Per = 0.0;
	
================================================
  
Array Declaration:
	
	DataType <ArrayName>[ArraySize];
	
Exa:

	int ID[15];
	
	char Name[40];	
	char DOB[20];
	
	float Distance[20];
